## Task 2 — Cinema Seats

```python
seats = [
    ["A", "A", "B", "A", "A"],
    ["A", "B", "A", "A", "A"],
    ["A", "A", "A", "B", "A"],
    ["B", "A", "A", "A", "A"]
]

print("Current seating plan:")
for row in seats:
    print(" ".join(row))

row = int(input("Enter row (0-3): "))
column = int(input("Enter column (0-4): "))

if 0 <= row < len(seats) and 0 <= column < len(seats[row]):
    if seats[row][column] == "A":
        seats[row][column] = "B"
        print("Seat booked successfully.")
    else:
        print("That seat is already booked.")
else:
    print("That position is outside the seating plan.")

print("Updated seating plan:")
for row in seats:
    print(" ".join(row))
```

## Task 3 — Rainfall Analysis

```python
rainfall = [
    [42, 38, 51, 47],
    [35, 41, 39, 44],
    [50, 48, 46, 53]
]

total = 0
number_of_values = 0
highest = rainfall[0][0]

for row in rainfall:
    for value in row:
        total += value
        number_of_values += 1
        if value > highest:
            highest = value

average = total / number_of_values
print("Total rainfall:", total)
print(f"Average rainfall: {average:.2f}")

for row in range(len(rainfall)):
    town_total = 0
    for column in range(len(rainfall[row])):
        town_total += rainfall[row][column]
    print(f"Town {row + 1} total: {town_total}")

print("Highest rainfall:", highest)
```

Expected results:

- total: `534`
- average: `44.50`
- town totals: `178`, `159`, `197`
- highest: `53`

## Task 4 — Searching a Warehouse

```python
products = [
    ["BK101", "PN205", "ER310"],
    ["NT420", "FL115", "ST225"],
    ["CP330", "MK440", "GL550"]
]

target = input("Enter a product code: ").upper()
found = False

for shelf in range(len(products)):
    for position in range(len(products[shelf])):
        if products[shelf][position].upper() == target:
            print(f"Found on shelf {shelf}, position {position}.")
            found = True

if not found:
    print("Product code not found.")
```

## Task 5 — Connect Three

```python
def display_board(board):
    for row in board:
        print(" ".join(row))


def has_won(board, player):
    directions = [
        (0, 1),   # horizontal
        (1, 0),   # vertical
        (1, 1),   # diagonal down and right
        (1, -1)   # diagonal down and left
    ]

    for row in range(len(board)):
        for column in range(len(board[row])):
            for row_change, column_change in directions:
                end_row = row + (2 * row_change)
                end_column = column + (2 * column_change)

                if 0 <= end_row < len(board) and 0 <= end_column < len(board[end_row]):
                    if (board[row][column] == player and
                            board[row + row_change][column + column_change] == player and
                            board[end_row][end_column] == player):
                        return True

    return False


board = [["-" for column in range(5)] for row in range(5)]
player = "X"
valid_moves = 0

while valid_moves < 25:
    display_board(board)
    print(f"Player {player}")
    row = int(input("Enter row (0-4): "))
    column = int(input("Enter column (0-4): "))

    if not (0 <= row < len(board) and 0 <= column < len(board[row])):
        print("That position is outside the board.")
    elif board[row][column] != "-":
        print("That position is already occupied.")
    else:
        board[row][column] = player
        valid_moves += 1

        if has_won(board, player):
            display_board(board)
            print(f"Player {player} wins!")
            break

        player = "O" if player == "X" else "X"
else:
    display_board(board)
    print("The board is full. The game is a draw.")
```

## Reflection — Example Using Task 2

The two-dimensional array is declared when `seats` is assigned a list containing four inner lists. An individual element is accessed using `seats[row][column]` and updated by assigning it `"B"`. The outer display loop selects one row at a time. `join()` processes the values in that row for output. Invalid indexes are prevented by checking that both indexes are at least zero and below the length of the appropriate list before accessing the element.
